\begin{frame}{Model: Setup}

\textbf{Metro area} $S$ partitioned into municipalities $g \in G$; fixed total population $N$.

\medskip
\begin{columns}[T]
\column{0.50\textwidth}
\begin{block}{Households}
  \small
  Cobb-Douglas over consumption \& housing; Fréchet taste shocks $v_{ij}(\omega)$.\\[4pt]
  Choose $(i,j)$; earn $w_j$, pay $q_i^R$, bear congestion disutility
  \[
    \phi(N_i,L_i) = \eta_N N_i^{1+\rho}+\eta_L L_i^{1+\rho}
  \]
  Location shares: $\;\pi_{ij}\propto T_i E_j\tilde{V}_{ij}^{\,\epsilon}$
\end{block}

\column{0.46\textwidth}
\begin{block}{Firms at $j$}
  \small
  CRS production; commercial land $\bar H_j^F$ set by zoning.
\end{block}

\vspace{0.25cm}
\begin{block}{Governments}
  \small
  \textbf{Municipality $g$:} sets $\bar H_i^R$ and $\bar H_i^F$\\[2pt]
  \textbf{Transport authority:} sets $\{I_{jk}\}$\\[2pt]
  $\Rightarrow$ Simultaneous Nash equilibrium
\end{block}
\end{columns}

\medskip
\small
\textbf{Zoning proxy:} housing supply elasticity $\hat\gamma_i$
from \citeauthor{Baum-Snow2024-ii}~(\citeyear{Baum-Snow2024-ii})
(lower $\hat\gamma_i$ = tighter zoning = smaller $\bar H_i^R$).

\end{frame}

%% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}[shrink=5]{Municipal Problem: FOC $\Rightarrow$ Propositions 1 \& 2}

Municipality $g$ maximizes $W^g = \sum_{i\in S^g} N_i\bigl(\bar U_i - \phi(N_i,L_i)\bigr)$.

\smallskip
\textbf{Interior FOC for residential capacity $\bar H_i^R$:}

{\small
\begin{align*}
&\underbrace{\bar U_i\,\frac{\partial N_i}{\partial \bar H_i^R}}_{\substack{(+)\\\text{in-movers}}}
+\;
N_i\underbrace{\frac{\partial \bar U_i}{\partial q_i^R}\frac{\partial q_i^R}{\partial \bar H_i^R}}_{\substack{(+)\\\text{rent falls}}}\\[8pt]
+\;&
N_i\underbrace{\frac{\partial \bar U_i}{\partial \phi}\frac{\partial \phi}{\partial N_i}\frac{\partial N_i}{\partial \bar H_i^R}}_{\substack{(-)\\\text{congestion cost}}}
+\;
N_i\underbrace{\left(-\frac{\partial \bar U_i}{\partial \phi}\frac{\partial \phi}{\partial L_i}\frac{\partial L_i}{\partial \bar H_i^F}\right)}_{\substack{(+)\\\text{less worker influx}}}
= 0
\end{align*}
}

\begin{columns}[T]
\column{0.50\textwidth}
\begin{alertblock}{Proposition 1}
  \small
  $\partial\phi/\partial L_i = (1+\rho)\eta_L L_i^{\rho}$ increases in $L_i$.\\[3pt]
  Higher $L_i$ $\Rightarrow$ larger \emph{worker-influx} cost in FOC $\Rightarrow$ municipality restricts residential zoning.\\[4pt]
  \textbf{Test:} $\;\partial\hat\gamma_i/\partial\log L_i < 0$
\end{alertblock}

\column{0.49\textwidth}
\begin{block}{Proposition 2}
  \small
  Transit $\Rightarrow$ higher $\tilde V_{ij}$ $\Rightarrow$ upward pressure on $N_i$.\\[3pt]
  Larger $N_i$ amplifies the \emph{congestion} term $\Rightarrow$ municipality preemptively tightens zoning.\\[4pt]
  \textbf{Test:} $\;\hat\gamma_i|_{\text{near transit}} < \hat\gamma_i|_{\text{far}}$
\end{block}
\end{columns}

\end{frame}

%% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}[shrink=5]{Transportation Authority: FOC $\Rightarrow$ Proposition 3}

Authority maximizes $N\cdot\Gamma\!\left(1-\tfrac{1}{\epsilon}\right)\Phi^{1/\epsilon}$
subject to $\sum_{jk}\delta_{jk}I_{jk}\leq K$.

\medskip
\textbf{Interior FOC for link $(j,k)$:}
\[
  \underbrace{\frac{N}{\epsilon}\,\Gamma\!\!\left(1-\tfrac{1}{\epsilon}\right)\Phi^{1/\epsilon-1}
  \frac{\partial\Phi}{\partial I_{jk}}}_{\text{Marginal Social Benefit (MSB)}}
  \;=\; \mu\,\delta_{jk}
\]

\smallskip
The MSB increases in ridership on link $(j,k)$:
\[
  \frac{\partial\Phi}{\partial I_{jk}}
  \;\propto\;
  \underbrace{\sum_{\text{users of }(j,k)} n_{ij}}_{\text{ridership}}
  \;\times\;
  \underbrace{\left|\frac{\partial\tau_{ij}}{\partial I_{jk}}\right|}_{\text{cost reduction}}
\]
Transit ridership $n_{ij}$ depends on the density of residents \emph{and} jobs near the corridor.

\begin{alertblock}{Proposition 3}
\[
  \underbrace{\text{Restrictive zoning}}_{\bar H_i^R\downarrow}
  \;\Longrightarrow\;
  \underbrace{N_i\downarrow,\; L_j\downarrow}_{\text{near transit}}
  \;\Longrightarrow\;
  \underbrace{n_{ij}\downarrow}_{\text{ridership}}
  \;\Longrightarrow\;
  \underbrace{\partial\Phi/\partial I_{jk}\,\downarrow}_{\text{MSB}}
  \;\Longrightarrow\;
  I^*_{jk}\downarrow
\]
\textbf{Test:} station-area density/ridership response to transit is weaker
where land-use capacity is lower.
\end{alertblock}

\end{frame}

%% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Overview: Three Proposition Tests}

\begin{block}{Empirical chain from the model's FOC}
\begin{enumerate}
  \item \textbf{P1.} High employment concentration $\Rightarrow$ tighter residential zoning
  \item \textbf{P2.} Transit access raises population pressure $\Rightarrow$ tighter zoning response
  \item \textbf{P3.} Restrictive land use $\Rightarrow$ lower station-area transit returns
\end{enumerate}
\end{block}

\vspace{0.3cm}
\textbf{Data}
\begin{itemize}
  \item Tract-level housing supply elasticity $\hat{\gamma}_i$: \citeauthor{Baum-Snow2024-ii} (\citeyear{Baum-Snow2024-ii}), Dallas metro (N\,=\,1,016 tracts)
  \item Employment inflows $L_i$: LODES 2023, aggregated to Census tract
  \item Transit access: DART station locations (148 stations), 0.5-mile buffer
  \item P3 data audit: existing NHGIS extract has tract population and housing units,
        but is not yet sufficient for a station-buffer panel
\end{itemize}

\vspace{0.2cm}
\small\textit{Tests are equilibrium consistency checks, not causal estimates.
              Each proposition maps to a model FOC.}
\end{frame}

\begin{frame}[shrink=3]{P1: Specification}

\textbf{Model prediction} (from municipal FOC, worker-influx term):
\[
  \frac{\partial\hat\gamma_i}{\partial\log L_i} < 0
\]
Higher employment at a tract raises the marginal disutility of additional workers,
so municipalities choose tighter residential zoning.

\bigskip
\textbf{OLS regression} \quad (Census tract, Dallas metro; $N = 1{,}016$):
\[
  \hat\gamma_i
  \;=\; \alpha
  + \underbrace{\beta_1}_{<\,0}\log(1+L_i)
  + \beta_2\,\text{pctdis}_i
  + \varepsilon_i
\]

\bigskip
{\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}lp{0.68\linewidth}@{}}
\toprule
Variable & Definition \\
\midrule
$\hat\gamma_i$        & Housing supply elasticity (Baum-Snow \& Han 2024);
                        \emph{lower = tighter zoning} (proxies $\bar H_i^R$ in the model) \\[4pt]
$\log(1+L_i)$         & Log employment inflows at tract $i$ from LODES 2023
                        (proxies $L_i$ in the model) \\[4pt]
$\text{pctdis}_i$     & Fractional distance from tract centroid to CBD;
                        controls for the urban density gradient \\
\bottomrule
\end{tabular}
}

\end{frame}

\begin{frame}{P2: Specification}

\[
  \hat\gamma_i\big|_{\text{near transit}} \;<\; \hat\gamma_i\big|_{\text{far}}

\begin{frame}[shrink=7]{P3: NHGIS Data Audit}

\textbf{Question.} Can the NHGIS files currently in
\texttt{/Volumes/ORICO/Proposal for SYP} support the requested station-area P3 test?

\medskip
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{What is available}
\small
\begin{itemize}
  \item Two tract-level NHGIS nominal time-series CSV extracts
  \item DFW county coverage: 2,856 integrated tract rows
  \item Key available measures: total population, total housing units,
        and travel-time-to-work bins
  \item Census years include 1990, 2000, 2010, and 2020
\end{itemize}
\end{block}

\column{0.50\textwidth}
\begin{alertblock}{Why this is not sufficient}
\small
\begin{itemize}
  \item No NHGIS GIS boundary shapefiles for 1990/2000/2010/2020
  \item Nominal tracts are not harmonized geographies; codes can refer to
        changed areas over time
  \item No block-group extract, limiting 0.5-mile station-buffer precision
  \item Missing controls/outcomes: income, tenure, vacancy, race/ethnicity,
        commute mode or public-transit share
\end{itemize}
\end{alertblock}
\end{columns}

\vspace{0.1cm}
\begin{alertblock}{Conclusion}
\small
The existing NHGIS data can help seed a tract-level panel, but it is not enough
to estimate P3 as specified in the project instructions.
\end{alertblock}

\end{frame}

\begin{frame}[shrink=5]{P3: Revised Station-Area Specification}

\textbf{Model prediction} (from transportation authority FOC):
restrictive land use lowers the density/ridership response that enters transit's
marginal social benefit.

\[
  Y_{sbt}
  =
  \alpha_s+\lambda_t
  + \beta_1 PostTransit_{st}
  + \underbrace{\beta_2}_{<\,0}
      (PostTransit_{st}\times TightZoning_b)
  + X_{bt}'\Gamma+\varepsilon_{sbt}
\]

\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lp{0.66\linewidth}@{}}
\toprule
Variable & Definition \\
\midrule
$s,b,t$ & Station, intersected buffer geography, and year \\[3pt]
$Y_{sbt}$ & Population density, housing-unit density, transit commute share,
or station ridership if available \\[3pt]
$PostTransit_{st}$ & Indicator for years after station $s$ opens \\[3pt]
$TightZoning_b$ & Direct zoning capacity, minimum lot size, single-family share,
or low pre-opening density/built-form proxy \\[3pt]
$X_{bt}$ & Income, race/ethnicity, tenure, vacancy, and CBD-distance controls \\
\bottomrule
\end{tabular}

\vspace{0.15cm}
\begin{alertblock}{Interpretation}
\small
This is the empirical object in the instructions: realized local development or
ridership response to transit, not a cross-metro network-length regression.
\end{alertblock}

\end{frame}

\begin{frame}[shrink=15]{Interpretation and Next Steps}

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{What the tests show}
\begin{itemize}\small
  \item \textbf{P1 \checkmark} Employment-concentrated tracts are systematically less permissive, consistent with the municipal congestion-cost FOC.
  \item \textbf{P2 \checkmark} DART-adjacent tracts are significantly less permissive than matched non-adjacent tracts, consistent with the transit-tightening channel.
  \item \textbf{P3 data gap} Current NHGIS files are not sufficient for the requested station-buffer event-study panel.
  \item \textbf{P3 interim} Existing 2020 tract evidence is consistent with weaker density near DART where land-use capacity is tighter, but it is not yet a causal return estimate.
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{alertblock}{What's actually needed for P3}
\begin{itemize}\small
  \item NHGIS tract or block-group GIS boundaries for 1990, 2000, 2010, and 2020
  \item NHGIS attributes for population, housing, tenure, vacancy, income, race/ethnicity, and commute mode
  \item Harmonization to a common 2010 or 2020 geography
  \item Station buffers joined to census geographies and station opening years
  \item Direct zoning, land-use capacity, or pre-opening built-form measures around stations
\end{itemize}
\end{alertblock}

\vspace{0.2cm}
\begin{block}{P2 refinements}
\begin{itemize}\small
  \item Expand to other metros in Baum-Snow sample
  \item Use distance to nearest station (continuous)
  \item Historical station-siting instrument for causal test
\end{itemize}
\end{block}
\end{columns}
